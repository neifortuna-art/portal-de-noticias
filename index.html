    <script>

        // Suas credenciais oficiais do Supabase

        const SUPABASE_URL = 'https://lrukpzgasxowmqkuwwcg.supabase.co';

        const SUPABASE_ANON_KEY = 'sb_publishable_jKoVdpcQ9QMjRA0oLzpd1Q_WlfBq1VZ';


        const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);


        async function carregarNoticias() {

            const { data, error } = await supabaseClient

                .from('noticias')

                .select('*')

                .order('created_at', { ascending: false });


            const container = document.getElementById('lista-noticias');

            

            if (error) {

                container.innerHTML = '<p style="color: red;">Erro ao carregar notícias. Verifique a conexão com o banco.</p>';

                console.error(error);

                return;

            }


            if (!data || data.length === 0) {

                container.innerHTML = '<p>Nenhuma notícia cadastrada no momento.</p>';

                return;

            }


            container.innerHTML = '';

            data.forEach(noticia => {

                const div = document.createElement('div');

                div.className = `noticia-card ${noticia.é_urgente ? 'urgente' : ''}`;

                

                div.innerHTML = `

                    ${noticia.é_urgente ? '<p style="color:red; font-weight:bold; margin:0 0 10px 0;">🚨 URGENTE</p>' : ''}

                    <h2>${noticia.titulo || 'Sem Título'}</h2>

                    <div class="meta">

                        Por ${noticia.autor || 'Redação'} | Categoria: ${noticia.editora || 'Geral'} 

                        | Publicado em: ${new Date(noticia.created_at).toLocaleDateString('pt-BR')}

                    </div>

                    ${noticia.imagem_url ? `<img src="${noticia.imagem_url}" alt="Capa da notícia">` : ''}

                    <p>${noticia.conteudo ? noticia.conteudo : ''}</p>

                `;

                container.appendChild(div);

            });

        }


        carregarNoticias();

    </script>
